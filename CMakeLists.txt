cmake_minimum_required(VERSION 3.20)
project(QingTou VERSION 0.9.2 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "QingTou is a Windows desktop application.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(capture_core src/image.cpp src/selection.cpp src/recorder.cpp)
target_include_directories(capture_core PUBLIC src)
target_compile_definitions(capture_core PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_link_libraries(capture_core PUBLIC d2d1 dwrite windowscodecs ole32 uuid gdi32 user32 shell32)
add_executable(QingTou WIN32 src/main.cpp src/theme.cpp resources/app.rc)
target_include_directories(QingTou PRIVATE resources)
target_link_libraries(QingTou PRIVATE capture_core comctl32 comdlg32 dwmapi advapi32)
set_target_properties(QingTou PROPERTIES
  RUNTIME_OUTPUT_DIRECTORY "${PROJECT_SOURCE_DIR}/dist"
  RUNTIME_OUTPUT_DIRECTORY_RELEASE "${PROJECT_SOURCE_DIR}/dist"
)
if(MINGW)
  target_compile_options(capture_core PRIVATE -Wall -Wextra -Wpedantic -finput-charset=UTF-8)
  target_compile_options(QingTou PRIVATE -Wall -Wextra -Wpedantic -finput-charset=UTF-8)
  target_link_options(QingTou PRIVATE -municode -static -static-libgcc -static-libstdc++ -s)
elseif(MSVC)
  target_compile_options(capture_core PUBLIC /utf-8 /W4 /permissive-)
  # app.rc already embeds the complete manifest, including DPI and UAC settings.
  target_link_options(QingTou PRIVATE /MANIFEST:NO)
endif()
include(CTest)
option(QINGTOU_ENABLE_DESKTOP_TESTS "Run tests that read an interactive Windows desktop" OFF)
if(BUILD_TESTING)
  add_executable(capture_tests tests/core_tests.cpp tests/wand_quality_tests.cpp tests/selection_session_tests.cpp)
  target_link_libraries(capture_tests PRIVATE capture_core)
  if(MINGW)
    target_link_options(capture_tests PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  add_test(NAME image_codecs_and_selection COMMAND capture_tests)
  add_executable(window_capture_tests tests/window_capture_tests.cpp src/theme.cpp resources/app.rc)
  target_include_directories(window_capture_tests PRIVATE resources)
  target_link_libraries(window_capture_tests PRIVATE capture_core comctl32 comdlg32 dwmapi advapi32)
  if(MINGW)
    target_compile_options(window_capture_tests PRIVATE -Wall -Wextra -Wpedantic -finput-charset=UTF-8)
    target_link_options(window_capture_tests PRIVATE -static -static-libgcc -static-libstdc++)
  elseif(MSVC)
    target_link_options(window_capture_tests PRIVATE /MANIFEST:NO)
  endif()
  add_test(NAME wand_click_and_edit COMMAND window_capture_tests --wand)
  if(QINGTOU_ENABLE_DESKTOP_TESTS)
    add_test(NAME desktop_capture_and_recording COMMAND capture_tests --desktop)
    set_tests_properties(desktop_capture_and_recording PROPERTIES LABELS desktop)
    add_test(NAME capture_window_visibility COMMAND window_capture_tests)
    set_tests_properties(capture_window_visibility PROPERTIES LABELS desktop RUN_SERIAL TRUE TIMEOUT 20)
  endif()
endif()
